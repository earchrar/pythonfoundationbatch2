    CREATE TABLE IF NOT EXISTS seats(
        id INT AUTO_INCREMENT PRIMARY KEY,
        busname VARCHAR(50),
        seatno VARCHAR(5),
        status ENUM('idle','occupied') DEFAULT 'idle',
        user_id BIGINT NULL,
        username VARCHAR(255),
        booked_at TIMESTAMP NULL
    );
